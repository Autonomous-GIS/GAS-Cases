import requests
import json
import geopandas as gpd
import pandas as pd

def download_data():
    # Define the endpoint and related configurations
    base_url = "https://api.census.gov/data/2021/acs/acs5"
    api_key = "918876e93cf30566c02b367fcad29644d861817e"
    dataset_year = "2021"
    dataset_source = f"ACS {dataset_year}"

    # Variables we need to fetch
    variables = [
        # B01001_001E: Total population
        "B01001_001E"
    ]

    # Fetch the variable labels
    response = requests.get(f"{base_url}/variables.json")
    variables_metadata = response.json()

    # Helper function to get variable labels
    def get_variable_label(var_name):
        label = variables_metadata['variables'][var_name]['label']
        return label.replace("Estimate!!", "").strip()

    # Construct the URL for the data request
    get_vars = ",".join(variables)
    url = f"{base_url}?get={get_vars},NAME&for=county:*&key={api_key}"

    # Download data from Census API
    response = requests.get(url)
    response.raise_for_status()
    data = response.json()

    # Create a DataFrame
    df = pd.DataFrame(data[1:], columns=[f"{var}:{get_variable_label(var)}" for var in variables] + ["Name", "State_FIPS", "County_FIPS"])
    df["Year"] = dataset_year
    df["Source"] = dataset_source

    # Convert DataFrame to GeoDataFrame
    df['County_FIPS'] = df['State_FIPS'] + df['County_FIPS']
    gdf = gpd.GeoDataFrame(df)

    # Save the final output as a geojson file
    output_path = "/home/abk6234/GAS/mysite2/Data/us_county_177105.geojson"
    gdf.to_file(output_path, driver='GeoJSON')

import pandas as pd
import geopandas as gpd

boundary_path = "Data/contiguous_counties_194557_226758.geojson"
population_path = "/home/abk6234/GAS/mysite2/Data/us_county_177105.geojson"
out_path = "Data/contiguous_counties_with_2021_population.geojson"

boundary = gpd.read_file(boundary_path).copy()
pop = gpd.read_file(population_path).copy()

boundary["county_id"] = boundary["GEOID"].astype(str).str.zfill(5)
pop["county_id"] = pop["County_FIPS"].astype(str).str.extract(r"(\d+)")[0].str.zfill(5)

pop = pop[pop["Year"].astype(str) == "2021"].copy()
pop["population_2021"] = pd.to_numeric(
    pop["B01001_001E:Total:"].astype(str).str.replace(",", "", regex=False),
    errors="coerce"
)

lower48_exclude = {"02", "15", "72", "60", "66", "69", "78"}
boundary = boundary[~boundary["county_id"].str[:2].isin(lower48_exclude)].copy()

pop_tab = pop[["county_id", "population_2021"]].drop_duplicates(subset=["county_id"])

result = boundary.merge(pop_tab, on="county_id", how="left")
result = gpd.GeoDataFrame(result, geometry=boundary.geometry.name, crs=boundary.crs)

result.to_file(out_path, driver="GeoJSON")

import geopandas as gpd
import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.colors as mcolors
from matplotlib.colorbar import ColorbarBase
import matplotlib
import numpy as np
import random
import mapclassify

# ── 1. CONFIG ──────────────────────────────────────────────────────────────────
DATA_PATH = "Data/county_population_join_616450_934497.geojson"
VALUE_COLUMN = "population_2021"
CLASSIFICATION = "quantiles"
K_CLASSES = 7

EDGE_COLOR = "#C0C0C0"
EDGE_WIDTH = 0.08
BG_COLOR = "#FFFFFF"
FIG_COLOR = "#FFFFFF"
TEXT_COLOR = "#000000"
TITLE = "2021 County Population Across the Contiguous United States"
DPI = 180

rand_num = random.randint(100000, 999999)
OUTPUT_FILE = f"/home/abk6234/GAS/mysite2/Output/County_Population_{rand_num}.png"

# ── 2. COLOR SETS ──────────────────────────────────────────────────────────────
COLOR_SETS = {
    "Blues": {"cmap": "Blues", "bar_label": "#185FA5", "spine": "#C8D8E8", "tick": "#111111"},
    "Greens": {"cmap": "Greens", "bar_label": "#3B6D11", "spine": "#C8E8D0", "tick": "#111111"},
    "Oranges": {"cmap": "Oranges", "bar_label": "#854F0B", "spine": "#F0D8C0", "tick": "#111111"},
}
chosen = COLOR_SETS["Blues"]
CMAP = chosen["cmap"]
BAR_LABEL = chosen["bar_label"]
SPINE_COL = chosen["spine"]
TICK_COL = chosen["tick"]

# ── 3. LOAD DATA ───────────────────────────────────────────────────────────────
gdf = gpd.read_file(DATA_PATH)

if "geometry" not in gdf.columns:
    raise ValueError("Dataset does not include a geometry column required for mapping.")

gdf = gdf.copy()
gdf[VALUE_COLUMN] = pd.to_numeric(gdf[VALUE_COLUMN], errors="coerce")

# Filter to contiguous U.S. by excluding Alaska, Hawaii, and territories
exclude_codes = {"02", "15", "60", "66", "69", "72", "78"}
if "STATEFP" in gdf.columns:
    gdf = gdf[~gdf["STATEFP"].astype(str).str.zfill(2).isin(exclude_codes)].copy()

# Reproject for cleaner national display if geographic
if gdf.crs is not None:
    try:
        if gdf.crs.is_geographic:
            gdf = gdf.to_crs(epsg=5070)
    except Exception:
        pass

# ── 4. AXIS LABELS ─────────────────────────────────────────────────────────────
crs = gdf.crs
if crs and crs.is_geographic:
    x_label = "Longitude (degrees)"
    y_label = "Latitude (degrees)"
else:
    try:
        axes = crs.axis_info
        x_label = f"{axes[0].name} ({axes[0].unit_name})"
        y_label = f"{axes[1].name} ({axes[1].unit_name})"
    except Exception:
        x_label = "X"
        y_label = "Y"

crs_epsg = crs.to_epsg() if crs else None
crs_str = f"EPSG:{crs_epsg}" if crs_epsg else "Unknown CRS"
SUBTITLE = f"CRS: {crs_str}"

# ── 5. VALUE PREP ──────────────────────────────────────────────────────────────
values = gdf[VALUE_COLUMN]
valid_mask = values.notna()
clean_values = values[valid_mask]

if clean_values.empty:
    raise ValueError(f"No valid numeric values in '{VALUE_COLUMN}'")

vmin, vmax = clean_values.min(), clean_values.max()

unique_count = clean_values.nunique()
k = min(K_CLASSES, unique_count)

if CLASSIFICATION == "natural_breaks":
    classifier = mapclassify.NaturalBreaks(clean_values, k=k)
elif CLASSIFICATION == "quantiles":
    classifier = mapclassify.Quantiles(clean_values, k=k)
elif CLASSIFICATION == "equal_interval":
    classifier = mapclassify.EqualInterval(clean_values, k=k)
elif CLASSIFICATION == "std_mean":
    classifier = mapclassify.StdMean(clean_values)
else:
    raise ValueError("Unsupported classification type")

discrete_cmap = matplotlib.colormaps.get_cmap(CMAP).resampled(classifier.k)

# Build color column safely as object dtype with hex colors
gdf["_color"] = pd.Series(["#D3D3D3"] * len(gdf), index=gdf.index, dtype="object")
class_colors = [mcolors.to_hex(discrete_cmap(int(cls))) for cls in classifier.yb]
gdf.loc[clean_values.index, "_color"] = class_colors

# ── 6. FIGURE ──────────────────────────────────────────────────────────────────
fig = plt.figure(figsize=(14, 9), facecolor=FIG_COLOR)

ax = fig.add_axes([0.06, 0.12, 0.72, 0.80])
ax.set_facecolor(BG_COLOR)

for spine in ax.spines.values():
    spine.set_edgecolor(SPINE_COL)
    spine.set_linewidth(0.6)

ax.tick_params(colors=TICK_COL, labelsize=9)
ax.set_xlabel(x_label, fontsize=10, fontweight="bold", color=TEXT_COLOR)
ax.set_ylabel(y_label, fontsize=10, fontweight="bold", color=TEXT_COLOR)

gdf.plot(
    ax=ax,
    color=gdf["_color"],
    edgecolor=EDGE_COLOR,
    linewidth=EDGE_WIDTH
)

# ── 7. TITLE ───────────────────────────────────────────────────────────────────
ax.set_title(TITLE, fontsize=15, fontweight="bold", loc="left", pad=12)
fig.text(0.06, 0.03, SUBTITLE, fontsize=10, fontweight="bold", color=TEXT_COLOR)

# ── 8. LEGEND OUTSIDE AXES ─────────────────────────────────────────────────────
cb_ax = fig.add_axes([0.82, 0.22, 0.03, 0.56])

bounds = classifier.bins
boundaries = np.concatenate(([vmin], bounds))
norm = mcolors.BoundaryNorm(boundaries, discrete_cmap.N)

cb = ColorbarBase(
    cb_ax,
    cmap=discrete_cmap,
    norm=norm,
    boundaries=boundaries,
    orientation="vertical"
)
cb_ax.set_title("Population\n2021\n(quantiles)", fontsize=9, pad=10, color=BAR_LABEL)
cb.outline.set_edgecolor(SPINE_COL)
cb.ax.tick_params(labelsize=8, colors=TICK_COL)

ax.set_aspect("equal")
ax.margins(0.01)

plt.savefig(OUTPUT_FILE, dpi=DPI, bbox_inches="tight", facecolor=FIG_COLOR)
print(f"__OUTPUT_PATH__={OUTPUT_FILE}")
print(f"__FEATURE_COUNT__={len(gdf)}")
